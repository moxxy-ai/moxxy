cmake_minimum_required(VERSION 3.24)
project(moxxy_computer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(geometry-test tests/geometry.cpp)
target_include_directories(geometry-test PRIVATE src)
add_test(NAME geometry COMMAND geometry-test)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    message(FATAL_ERROR "Computer Use supports Windows x64 only")
  endif()
  add_executable(moxxy-computer src/main.cpp src/desktop.cpp src/capture.cpp src/input.cpp src/control.cpp src/input-guard.cpp src/control-panel.cpp src/app-catalog.cpp src/text-actions.cpp)
  target_compile_definitions(moxxy-computer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(moxxy-computer PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET moxxy-computer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(moxxy-computer PRIVATE windowsapp uiautomationcore user32 gdi32 ole32 oleaut32 dwmapi d3d11 dxgi windowscodecs crypt32 shell32 propsys)
  target_link_options(moxxy-computer PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
  install(TARGETS moxxy-computer RUNTIME DESTINATION bin/win32-x64)
  add_executable(moxxy-computer-fixture WIN32 tests/fixture.cpp)
  target_include_directories(moxxy-computer-fixture PRIVATE src)
  target_compile_definitions(moxxy-computer-fixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(moxxy-computer-fixture PRIVATE /EHsc /utf-8)
  set_property(TARGET moxxy-computer-fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(moxxy-computer-fixture PRIVATE windowsapp user32 gdi32 ole32 oleaut32 shell32 uiautomationcore)
  target_link_options(moxxy-computer-fixture PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
endif()
